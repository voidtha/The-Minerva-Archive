-- Helper Functions
local function getHRP()
    local char = LocalPlayer.Character or LocalPlayer.CharacterAdded:Wait()
    return char:WaitForChild("HumanoidRootPart", 5)
end

-- Flight Engine Logic
local function startFlyEngine()
    if flyConnection then flyConnection:Disconnect() end

    flyConnection = RunService.Heartbeat:Connect(function()
        local char = LocalPlayer.Character
        local hum = char and char:FindFirstChildWhichIsA("Humanoid")
        local rootPart = char and char:FindFirstChild("HumanoidRootPart")

        if flying and char and hum and rootPart then
            hum.PlatformStand = true
            
            local moveDirection = Vector3.zero
            if UserInputService:IsKeyDown(Enum.KeyCode.W) then moveDirection += rootPart.CFrame.LookVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.S) then moveDirection -= rootPart.CFrame.LookVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.A) then moveDirection -= rootPart.CFrame.RightVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.D) then moveDirection += rootPart.CFrame.RightVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.Space) then moveDirection += Vector3.new(0, 1, 0) end
            if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) then moveDirection -= Vector3.new(0, 1, 0) end

            rootPart.AssemblyLinearVelocity = Vector3.zero
            rootPart.AssemblyAngularVelocity = Vector3.zero
            rootPart.CFrame = rootPart.CFrame + (moveDirection * flySpeed * 0.1)
        end
    end)
end

startFlyEngine()

-- Manual Fly Key Toggle (F Key)
flyInputConnection = UserInputService.InputBegan:Connect(function(input, gameProcessed)
    if gameProcessed then return end
    if input.KeyCode == Enum.KeyCode.F then
        flying = not flying
        local char = LocalPlayer.Character
        local hum = char and char:FindFirstChildWhichIsA("Humanoid")
        if hum then
            hum.PlatformStand = flying
            if not flying then
                hum:ChangeState(Enum.HumanoidStateType.GettingUp)
            end
        end
    end
end)

-- Bypass Steal Logic
local isTeleporting = false
local loopConnection = nil
local canUseSteal = true

local function startTeleporting()
    local targetPart = Workspace:FindFirstChild(TARGET_PART_NAME, true)
    if not targetPart or not targetPart:IsA("BasePart") then return end
    targetPart.CanCollide = false
    
    loopConnection = RunService.Heartbeat:Connect(function()
        local hrp = getHRP()
        if not isTeleporting or not hrp or not hrp.Parent then return end
        targetPart.CFrame = hrp.CFrame * CFrame.new(LEG_OFFSET)
        hrp.CFrame = BYPASS_TARGET_CFRAME + TELEPORT_HEIGHT_OFFSET
        targetPart.CFrame = BYPASS_TARGET_CFRAME
    end)
end

local function stopTeleporting()
    if loopConnection then 
        loopConnection:Disconnect() 
        loopConnection = nil 
    end
    local hrp = getHRP()
    if hrp then
        hrp.AssemblyLinearVelocity = Vector3.zero
        hrp.AssemblyAngularVelocity = Vector3.zero
    end
end

ToggleButton.MouseButton1Click:Connect(function()
    if not canUseSteal and not isTeleporting then return end
    isTeleporting = not isTeleporting
    if isTeleporting then
        canUseSteal = false
        ToggleButton.Text = "Bypass Steal: ON"
        ToggleButton.TextColor3 = Color3.fromRGB(255, 255, 255)
        startTeleporting()
    else
        ToggleButton.Text = "Bypass Steal: OFF"
        ToggleButton.TextColor3 = Color3.fromRGB(150, 150, 150)
        stopTeleporting()
        task.spawn(function()
            for i = 15, 1, -1 do
                if isTeleporting then return end
                ToggleButton.Text = "Wait: " .. string.format("%.1f", i / 10) .. "s"
                task.wait(0.1)
            end
            if not isTeleporting then
                ToggleButton.Text = "Bypass Steal: OFF"
                canUseSteal = true
            end
        end)
    end
end)

-- Smart Server Hop Logic
local function smartServerHop()
    HopButton.Text = "Searching..."

    local placeId = game.PlaceId
    local currentJobId = game.JobId
    local req = (syn and syn.request) or (http and http.request) or http_request or request

    if not req then
        HopButton.Text = "Executor Unsupported"
        task.wait(2)
        HopButton.Text = "Server Hop: Low"
        return
    end

    local success, result = pcall(function()
        local url = string.format("https://games.roblox.com/v1/games/%d/servers/0?sortOrder=Asc&limit=100", placeId)
        local response = req({ Url = url, Method = "GET" })
        return HttpService:JSONDecode(response.Body)
    end)

    if success and result and result.data then
        local validServers = {}
        for _, server in ipairs(result.data) do
            if server.id ~= currentJobId and server.playing < server.maxPlayers and server.playing > 0 then
                table.insert(validServers, server)
            end
        end

        table.sort(validServers, function(a, b)
            return a.playing < b.playing
        end)

        if #validServers > 0 then
            HopButton.Text = "Teleporting (" .. validServers[1].playing .. " players)"
            TeleportService:TeleportToPlaceInstance(placeId, validServers[1].id, LocalPlayer)
        else
            HopButton.Text = "No Server Found"
            task.wait(2)
            HopButton.Text = "Server Hop: Low"
        end
    end
end

HopButton.MouseButton1Click:Connect(function()
    smartServerHop()
end)

-- Dragging Mechanics
local dragStd, dragStartStd, startPosStd
local function enableDragStd(guiObject)
    guiObject.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragStd = true
            dragStartStd = input.Position
            startPosStd = MainFrame.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then 
                    dragStd = false 
                end
            end)
        end
    end)
end

enableDragStd(HeaderLabel)
enableDragStd(DropdownButton)

UserInputService.InputChanged:Connect(function(input)
    if dragStd and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
        local delta = input.Position - dragStartStd
        MainFrame.Position = UDim2.new(startPosStd.X.Scale, startPosStd.X.Offset + delta.X, startPosStd.Y.Scale, startPosStd.Y.Offset + delta.Y)
    end
end)

charAddedConnection = LocalPlayer.CharacterAdded:Connect(function()
    stopTeleporting()
    task.wait(1)
    startFlyEngine()
end)
